versionSql.VIA_INGRESO.fecha	CASE WHEN (SELECT MAX(e.id) FROM stat_event e WHERE e.communication_id=c.id AND e.record_type='COM' AND e.record_id=c.id AND e.variable='VIA_INGRESO' AND e.event IN ('SET','STATUS') AND e.recorded_ms<=1234567890123) IS NULL THEN (SELECT MAX(dv.id) FROM stat_catalog_version dv JOIN stat_catalog_item di ON di.version_id=dv.id WHERE dv.catalog='VIA_INGRESO' AND di.selectable=TRUE AND dv.activated_ms<=1234567890123 AND (dv.retired_ms IS NULL OR dv.retired_ms>1234567890123) AND di.code=(CASE c.channel WHEN 'FORMULARIO_WEB_PUBLICO' THEN 'FORMULARIO_PUBLICO_ADULTO' WHEN 'FORMULARIO_WEB_NNA' THEN 'FORMULARIO_PUBLICO_NNA' WHEN 'PRESENCIAL' THEN 'PRESENCIAL' WHEN 'TELEFONO' THEN 'TELEFONO' WHEN 'CORREO' THEN 'CORREO' WHEN 'OFICIO' THEN 'OFICIO' WHEN 'REDES_SOCIALES' THEN 'REDES_SOCIALES' WHEN 'INTEGRANTE_DEFENSORIA' THEN 'INTEGRANTE_DEFENSORIA' WHEN 'OTRO' THEN 'OTRO' END) AND c.created_ms<=1234567890123) ELSE (SELECT e2.catalog_version_id FROM stat_event e2 WHERE e2.id=(SELECT MAX(e.id) FROM stat_event e WHERE e.communication_id=c.id AND e.record_type='COM' AND e.record_id=c.id AND e.variable='VIA_INGRESO' AND e.event IN ('SET','STATUS') AND e.recorded_ms<=1234567890123)) END
versionSql.VIA_INGRESO.vigente	CASE WHEN (SELECT MAX(e.id) FROM stat_event e WHERE e.communication_id=c.id AND e.record_type='COM' AND e.record_id=c.id AND e.variable='VIA_INGRESO' AND e.event IN ('SET','STATUS')) IS NULL THEN (SELECT MAX(dv.id) FROM stat_catalog_version dv JOIN stat_catalog_item di ON di.version_id=dv.id WHERE dv.catalog='VIA_INGRESO' AND di.selectable=TRUE AND dv.`current_catalog`=dv.catalog AND di.code=(CASE c.channel WHEN 'FORMULARIO_WEB_PUBLICO' THEN 'FORMULARIO_PUBLICO_ADULTO' WHEN 'FORMULARIO_WEB_NNA' THEN 'FORMULARIO_PUBLICO_NNA' WHEN 'PRESENCIAL' THEN 'PRESENCIAL' WHEN 'TELEFONO' THEN 'TELEFONO' WHEN 'CORREO' THEN 'CORREO' WHEN 'OFICIO' THEN 'OFICIO' WHEN 'REDES_SOCIALES' THEN 'REDES_SOCIALES' WHEN 'INTEGRANTE_DEFENSORIA' THEN 'INTEGRANTE_DEFENSORIA' WHEN 'OTRO' THEN 'OTRO' END)) ELSE (SELECT e2.catalog_version_id FROM stat_event e2 WHERE e2.id=(SELECT MAX(e.id) FROM stat_event e WHERE e.communication_id=c.id AND e.record_type='COM' AND e.record_id=c.id AND e.variable='VIA_INGRESO' AND e.event IN ('SET','STATUS'))) END
versionSql.VIGENTE	(SELECT d.id FROM stat_declared_version d WHERE d.communication_id=c.id ORDER BY d.registered_ms DESC,d.id DESC LIMIT 1)
